// Simple rules: only logged-in users can read/write everything.
// (Fine for a hackathon/demo; this avoids "permission denied" problems.)
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Optional: for the public room page without login (lite.html)
    match /liteRooms/{room}/messages/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['name','text','cid','createdAt'])
                    && request.resource.data.name is string && request.resource.data.name.size() <= 24
                    && request.resource.data.text is string && request.resource.data.text.size() > 0
                    && request.resource.data.text.size() <= 1000
                    && request.resource.data.cid is string;
      allow update, delete: if false;
    }

    // Everything else (users, contacts, chats, groups, rooms, status)
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
